package main

import (
 "context"
 "encoding/json"
 "fmt"
 "io"
 "os"
 "path/filepath"

 "runtime/pprof"
  "time"

 "github.com/steamwo/fssql/pkg/gosql"
)

// sample 保存本机公共 API 诊断样本
 type sample struct {
 Case string `json:"case"` // Case 是查询场景
 Iteration int `json:"iteration"` // Iteration 是预热后的样本号
 Rows int `json:"rows"` // Rows 是完整消费行数
 Seconds float64 `json:"seconds"` // Seconds 是查询及完整消费耗时
 Allocated uint64 `json:"allocated_bytes"` // Allocated 是本次 Go 分配字节数
 Allocs uint64 `json:"allocs"` // Allocs 是本次 Go 分配次数
 GC uint32 `json:"gc_cycles"` // GC 是本次垃圾回收次数
 ManagedAfter int64 `json:"managed_after"` // ManagedAfter 是结束后的受管占用
 }

// auditReader 生成确定性有界测试行
 type auditReader struct { next int // next 是下一行编号
 }

// Next 填充下一批合成数据
func (r *auditReader) Next(ctx context.Context,w gosql.BatchWriter)(int,error){
 count:=min(w.Capacity(),100000-r.next)
 for row:=0;row<count;row++ {
  id:=r.next+row
  for _,err:=range []error{w.SetInt64(row,0,int64(id)),w.SetString(row,1,[]string{"A","B","C","D","E"}[id%5]),w.SetString(row,2,"Example-张小明-AbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbcAbc")}{if err!=nil{return 0,err}}
 }
 r.next+=count
 if r.next>=100000{return count,io.EOF};return count,nil
}

// Close 释放本夹具的逻辑资源
func(r *auditReader)Close()error{return nil}

// main 输出只读分析结果，不修改引擎源文件
func main(){if err:=run();err!=nil{fmt.Fprintln(os.Stderr,err);os.Exit(1)}}

// run 复现 CSV 边界并采集完整消费的诊断样本
func run()error{
 root,err:=filepath.Abs(os.Args[1]);if err!=nil{return err}
 ctx:=context.Background()
 path:=filepath.Join(root,"quoted-empty.csv")
 if err:=os.WriteFile(path,[]byte("v\n1\n\"\"\n"),0600);err!=nil{return err}
 uri,err:=gosql.FileURI(path);if err!=nil{return err}
 sqlDB,err:=gosql.OpenSQLDB(gosql.Config{LocalFiles:&gosql.LocalFileConfig{RootDir:root},TempDir:filepath.Join(root,"csv-temp"),ManifestDir:filepath.Join(root,"csv-manifests")});if err!=nil{return err}
 for _,explicit:=range []bool{false,true} {
  options:="";if explicit{options=",columns={'v':'VARCHAR'}"}
  query:="SELECT v FROM read_csv('"+uri+"',header=true,allow_quoted_nulls=false"+options+")"
  rows,queryErr:=sqlDB.QueryContext(ctx,query)
  var values []any
  if queryErr==nil{
   for rows.Next(){var value any;if err:=rows.Scan(&value);err!=nil{queryErr=err;break};values=append(values,value)}
   if queryErr==nil{queryErr=rows.Err()};rows.Close()
  }
  message:="";if queryErr!=nil{message=queryErr.Error()}
  if err:=json.NewEncoder(os.Stdout).Encode(map[string]any{"case":"csv_quoted_empty","explicit_schema":explicit,"values":values,"error":message});err!=nil{return err}
 }
 sqlDB.Close()
 hmac,err:=gosql.NewHMACSHA256ScalarFunction([]byte("public-audit-test-key"));if err!=nil{return err}
 db,err:=gosql.Open(gosql.Config{TableFunctions:[]gosql.TableFunction{{Name:"audit_rows",Columns:[]gosql.Column{{Name:"id",Type:gosql.Type{ID:"BIGINT"}},{Name:"grp",Type:gosql.Type{ID:"VARCHAR"}},{Name:"txt",Type:gosql.Type{ID:"VARCHAR"}}},Open:func(context.Context)(gosql.BatchReader,error){return &auditReader{},nil}}},ScalarFunctions:[]gosql.ScalarFunction{hmac},MemoryLimitBytes:256<<20,SpillMemoryBudgetBytes:64<<20,TempDir:filepath.Join(root,"bench-temp"),ManifestDir:filepath.Join(root,"bench-manifests"),Workers:1});if err!=nil{return err}
 defer db.Close()
 cases:=[]struct{name,query string}{{"projection","SELECT txt FROM audit_rows()"},{"lower","SELECT LOWER(txt) FROM audit_rows()"},{"hmac","SELECT data_plane_hmac_sha256(txt) FROM audit_rows()"},{"sort","SELECT id FROM audit_rows() ORDER BY grp,id"},{"window","SELECT id,ROW_NUMBER() OVER (PARTITION BY grp ORDER BY id) FROM audit_rows()"}}
 if len(os.Args)>2 { filtered:=cases[:0];for _,scene:=range cases{if scene.name==os.Args[2]{filtered=append(filtered,scene)}};cases=filtered }
 for iteration:=-2;iteration<7;iteration++ {
  for _,scene:=range cases {
   runtime.GC()
   var before,after runtime.MemStats
   runtime.ReadMemStats(&before)
   rows:=0;start:=time.Now()
   _,err:=db.QueryChunks(ctx,scene.query,func(c *gosql.Chunk)error{rows+=c.RowCount();return nil})
   seconds:=time.Since(start).Seconds()
   runtime.ReadMemStats(&after)
   if err!=nil{return fmt.Errorf("%s: %w",scene.name,err)}
   if rows!=100000{return fmt.Errorf("%s rows=%d",scene.name,rows)}
   if iteration>=0{if err:=json.NewEncoder(os.Stdout).Encode(sample{Case:scene.name,Iteration:iteration,Rows:rows,Seconds:seconds,Allocated:after.TotalAlloc-before.TotalAlloc,Allocs:after.Mallocs-before.Mallocs,GC:after.NumGC-before.NumGC,ManagedAfter:db.MemoryUsed()});err!=nil{return err}}
  }
 }
 if len(os.Args)>2 {
  path:=filepath.Join(root,os.Args[2]+"-isolated-alloc.pprof")
  file,err:=os.Create(path);if err!=nil{return err}
  runtime.GC()
  if err:=pprof.Lookup("allocs").WriteTo(file,0);err!=nil{file.Close();return err}
  file.Close()
 }
 return nil
}
